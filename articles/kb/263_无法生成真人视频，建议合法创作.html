import bpy
from math import sin, radians

def foot_animation(fps=24, amplitude=5):
    scene = bpy.context.scene
    bone = bpy.data.objects['Armature'].pose.bones['foot_bone']
    for frame in range(250):
        scene.frame_set(frame)
        bone.rotation_euler.z = radians(amplitude * sin(frame/fps))
        bone.keyframe_insert(data_path="rotation_euler")

foot_animation()
