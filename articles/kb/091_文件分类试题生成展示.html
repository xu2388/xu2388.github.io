                if (!c) return;
                const rect = c.parentElement.getBoundingClientRect();
                const w = c.width || 400;

                ctx2.clearRect(0, 0, w, h);
                ctx2.fillStyle = bg;
                ctx2.fillRect(0, 0, w, h);

                // 网格
                ctx2.strokeStyle = grid;
                ctx2.lineWidth = 0.5;
                for (let i = 0; i <= 10; i++) {
                    let x = (i / 10) * w;
                    let y = (i / 10) * h;
                    ctx2.beginPath();
                    ctx2.moveTo(x, 0);
                    ctx2.lineTo(x, h);
                    ctx2.stroke();
                    ctx2.beginPath();
                    ctx2.moveTo(0, y);
                    ctx2.lineTo(w, y);
                    ctx2.stroke();
                }

                // 坐标轴
                ctx2.strokeStyle = text;
                ctx2.lineWidth = 1.2;
                let originX = w * 0.5;
                let originY = h * 0.85;
                ctx2.beginPath();
                ctx2.moveTo(0, originY);
                ctx2.lineTo(w, originY);
                ctx2.stroke();
                ctx2.beginPath();
                ctx2.moveTo(originX, 0);
                ctx2.lineTo(originX, h);
                ctx2.stroke();

                // 绘制函数曲线
                let points = [];
                let step = 0.02;
                let range = 6;
                for (let x = -range; x <= range; x += step) {
                    let val = func(x, alpha);
                    let px = originX + (x / range) * (w * 0.45);
                    let py = originY - (val / 1.2) * (h * 0.70);
                    points.push({ px, py, val });
                }

                ctx2.beginPath();
                ctx2.strokeStyle = color || '#6366f1';
                ctx2.lineWidth = 2.5;
                for (let i = 0; i < points.length; i++) {
                    let p = points[i];
                    if (i === 0) ctx2.moveTo(p.px, p.py);
                    else ctx2.lineTo(p.px, p.py);
                }
                ctx2.stroke();

                // 标记当前值 (如果alpha对应)
                let curX = 0.5;
                let curVal = func(curX, alpha);
                let cpx = originX + (curX / range) * (w * 0.45);
                let cpy = originY - (curVal / 1.2) * (h * 0.70);
                ctx2.beginPath();
                ctx2.arc(cpx, cpy, 5, 0, 2 * Math.PI);
                ctx2.fillStyle = '#fbbf24';
                ctx2.fill();
                ctx2.strokeStyle = '#fbbf24';
                ctx2.lineWidth = 2;
                ctx2.stroke();

                ctx2.font = '10px monospace';
